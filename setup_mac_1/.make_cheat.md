# General structures

    targets: dependencies
	    commands

    target: normal-dependencies | order-only-dependencies
	    commands

# Special variables

    $@ : variable containing the target
    $? : variable containing all dependencies newer than the target
    $^ : variable containing all dependencies
    $< : variable containing first dependency

# Equalities

    := : calculated once when the variable is defined
    = : calculated every time the variable is used

# Special functions

    $(wildcard d/a*) : if the directory d contains files ax, by, az, then this returns `ax az`
    $(patsubst a%, b%, ax ay az) : returns `bx by bz`
    $(foreach x, a b c, $(x)y) : returns `ay by cy`
    $(shell x) : returns the output of the shell-command x
